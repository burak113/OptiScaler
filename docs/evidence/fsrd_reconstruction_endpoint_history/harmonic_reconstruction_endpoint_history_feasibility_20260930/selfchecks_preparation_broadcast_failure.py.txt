from pathlib import Path
import hashlib,json,warnings
import numpy as np
import model
HERE=Path(__file__).resolve().parent
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
desc=dict(frequencies=[[3.23,1.17]],chosen=[0,1,2],groups=[[1,2]],original_groups=[[1,2]])
n,h,w=24,20,32;proj=model.physical_projection(desc,h,w);phi=proj['Phi']
ctrl=np.zeros((n,3));ctrl[0,0]=1;active=np.ones(n,bool)
raw=np.broadcast_to(.3+.012*phi[None,...,:1],(n,h,w,3)).copy()
def metadata(theta=0.,used=False):return {'frames':[dict(frame=i,epoch_start=0,atom_diagnostics=[dict(phase_increment=theta,phase_used=used,innovation=False)])for i in range(n)]}
def invoke(R0,D,source=raw,control=ctrl,mask=active,diag=None,Poverride=None,TPoverride=None,exposure=None,descriptors=None):
    P=source.copy()if Poverride is None else Poverride
    TP=P-D if TPoverride is None else TPoverride
    B=R0-(P-TP)
    return (*model.make_reconstruction_endpoint_history(source,P,TP,B,mask,control,metadata()if diag is None else diag,[desc]*n if descriptors is None else descriptors,exposure),B)
def component(beta):return .3+np.einsum('hwk,nkc->nhwc',phi,beta)
check_results=[]
def passed(name,**detail):check_results.append({'name':name,'passed':True,**detail})
weight_report=[]
for count in range(1,17):
    weights=model.endpoint_weights(count);time=np.arange(count,dtype=float)
    np.testing.assert_allclose(weights.sum(),1,atol=4e-16,rtol=0)
    np.testing.assert_allclose(weights@time,count-1,atol=4e-15,rtol=0)
    X=np.stack([np.ones(count),time],axis=1)
    expected=np.ones(1)if count==1 else np.array([1.,count-1])@np.linalg.pinv(X)
    np.testing.assert_allclose(weights,expected,atol=4e-15,rtol=0)
    gain=float(weights@weights);formula=1 if count==1 else 2*(2*count-1)/(count*(count+1))
    np.testing.assert_allclose(gain,formula,atol=4e-16,rtol=0)
    weight_report.append({'n':count,'weights':weights.tolist(),'IID_unit_noise_gain':gain,'weight_sum':float(weights.sum()),'linear_moment':float(weights@time)})
passed('OLS_endpoint_affine_invariants_and_closed_noise_gain',n16_gain=weight_report[-1]['IID_unit_noise_gain'])
beta=np.zeros((n,2,3));beta[:,0]=.006;beta[:,1]=.001
R0=component(beta);D=np.einsum('hwk,nkc->nhwc',phi,np.random.default_rng(401).normal(0,.001,(n,2,3)))
out,diag,C,B=invoke(R0,D)
np.testing.assert_allclose(C,R0,atol=2e-16,rtol=0)
assert C[0].tobytes()==(B+(raw-(raw-D)))[0].tobytes()
passed('endogenous_B_D_cancellation_constant_reconstruction_preserved',preDC_max_error=float(abs(C-R0).max()))
linear=beta+np.arange(n)[:,None,None]*np.array([[.00005,.00003,-.00002],[.00002,-.00003,.00004]])
Rlinear=component(linear);o,d,C,B=invoke(Rlinear,D)
np.testing.assert_allclose(C,Rlinear,atol=3e-16,rtol=0)
passed('linear_reconstruction_endpoint_preserved_in_transported_frame',preDC_max_error=float(abs(C-Rlinear).max()))

np.testing.assert_allclose(mc,Rmoving,atol=3e-16,rtol=0)
passed('constant_speed_co_moving_reconstruction_phase_preserved')
# Honest failure: a moving source phase does not imply static reconstruction phase.
source=component(moving);staticD=np.broadcast_to(.002*phi[None,...,:1],raw.shape).copy()
_,badphase,badc,_=invoke(R0,staticD,source=source,diag=metadata(.07,True))
countererror=float(abs(badc-R0).max());assert countererror>1e-5

assert idg['frames'][8]['history_frames']==[8] and idg['frames'][8]['history_count']==1

cuts=ctrl.copy();cuts[12,0]=1;cuts[18:,1]=.25
_,cd,cc,_=invoke(Rlinear,D,control=cuts)
assert cd['frames'][12]['history_frames']==[12] and cd['frames'][18]['history_frames']==[18]
